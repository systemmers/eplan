📚 Integration Guide

Step 1: Copy HTML Structure

base.html의 기존 navbar를 이 구조로 교체하세요.

Step 2: Add CSS

위의 style 태그 내용을 별도 CSS 파일로 추출하거나 header.css에 추가하세요.

Step 3: Add JavaScript

아래 JavaScript를 main.js에 추가하세요.

// React-style Navigation Menu document.addEventListener('DOMContentLoaded', function() { const menuItems = document.querySelectorAll('.menu-item[data-dropdown]'); let activeDropdown = null; let hoverTimeout = null; menuItems.forEach(item => { const dropdown = item.querySelector('.dropdown-container'); // Mouse enter item.addEventListener('mouseenter', function() { clearTimeout(hoverTimeout); // Close other dropdowns if (activeDropdown && activeDropdown !== dropdown) { activeDropdown.classList.remove('show'); } // Show current dropdown with slight delay hoverTimeout = setTimeout(() => { dropdown.classList.add('show'); activeDropdown = dropdown; item.classList.add('active'); }, 100); }); // Mouse leave item.addEventListener('mouseleave', function() { clearTimeout(hoverTimeout); hoverTimeout = setTimeout(() => { dropdown.classList.remove('show'); item.classList.remove('active'); if (activeDropdown === dropdown) { activeDropdown = null; } }, 150); }); // Keyboard navigation const link = item.querySelector('> a'); link.addEventListener('keydown', function(e) { if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); dropdown.classList.toggle('show'); item.classList.toggle('active'); } if (e.key === 'Escape') { dropdown.classList.remove('show'); item.classList.remove('active'); } }); }); // Close dropdowns on outside click document.addEventListener('click', function(e) { if (!e.target.closest('.menu-item')) { menuItems.forEach(item => { item.querySelector('.dropdown-container').classList.remove('show'); item.classList.remove('active'); }); activeDropdown = null; } }); });
